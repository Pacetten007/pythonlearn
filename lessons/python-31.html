1. Встроенные функции Python

Python имеет три основные функции для работы с системами счисления: bin(), oct(), hex() и универсальную функцию int().

Перевод из десятичной в другие системы:

# Двоичная система (основание 2)
decimal = 42
binary = bin(decimal)
print(binary)  # 0b101010
print(bin(42)[2:])  # 101010 (без префикса)

# Восьмеричная система (основание 8)
octal = oct(decimal)
print(octal)  # 0o52
print(oct(42)[2:])  # 52 (без префикса)

# Шестнадцатеричная система (основание 16)
hexadecimal = hex(decimal)
print(hexadecimal)  # 0x2a
print(hex(42)[2:])  # 2a (без префикса)
print(hex(42)[2:].upper())  # 2A (заглавные буквы)

Перевод из других систем в десятичную:

# Функция int() с указанием основания
# int(строка, основание)

# Из двоичной
binary_str = "101010"
decimal = int(binary_str, 2)
print(decimal)  # 42

# Из восьмеричной
octal_str = "52"
decimal = int(octal_str, 8)
print(decimal)  # 42

# Из шестнадцатеричной
hex_str = "2A"
decimal = int(hex_str, 16)
print(decimal)  # 42

# Регистр букв не важен
print(int("2a", 16))  # 42
print(int("2A", 16))  # 42

2. Форматирование чисел

Функция format() позволяет форматировать числа в различных системах счисления с дополнительными опциями.

Базовое форматирование:

number = 42

# Двоичная система
print(format(number, 'b'))   # 101010
print(format(number, '08b')) # 00101010 (8 бит с ведущими нулями)
print(format(number, '016b')) # 0000000000101010 (16 бит)

# Восьмеричная система
print(format(number, 'o'))   # 52
print(format(number, '04o'))  # 0052 (4 цифры с ведущими нулями)

# Шестнадцатеричная система
print(format(number, 'x'))   # 2a (строчные буквы)
print(format(number, 'X'))   # 2A (заглавные буквы)
print(format(number, '04x'))  # 002a (4 цифры, строчные)
print(format(number, '04X'))  # 002A (4 цифры, заглавные)

Форматирование с префиксами:

number = 42

# С префиксами
print(format(number, '#b'))  # 0b101010
print(format(number, '#o'))   # 0o52
print(format(number, '#x'))   # 0x2a
print(format(number, '#X'))   # 0x2A

# Комбинация префикса и ширины
print(format(number, '#08b')) # 0b101010 (минимум 8 символов)
print(format(number, '#04x'))  # 0x2a (минимум 4 символа)

3. Создание собственных функций

Можно создать универсальные функции для работы с любыми системами счисления.

Универсальная функция перевода в любую систему:

def decimal_to_base(n, base):
    """Переводит десятичное число в систему с основанием base"""
    if n == 0:
        return "0"
    
    digits = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    result = ""
    
    while n > 0:
        result = digits[n % base] + result
        n //= base
    
    return result

# Примеры
print(decimal_to_base(42, 2))   # 101010
print(decimal_to_base(42, 8))   # 52
print(decimal_to_base(42, 16))  # 2A
print(decimal_to_base(100, 5))  # 400 (пятеричная система)
print(decimal_to_base(255, 36)) # 73 (тридцатишестиричная система)

Универсальная функция перевода из любой системы:

def base_to_decimal(number_str, base):
    """Переводит число из системы с основанием base в десятичную"""
    digits = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    number_str = number_str.upper()  # Приводим к верхнему регистру
    
    decimal = 0
    for digit in number_str:
        decimal = decimal * base + digits.index(digit)
    
    return decimal

# Примеры
print(base_to_decimal("101010", 2))  # 42
print(base_to_decimal("52", 8))      # 42
print(base_to_decimal("2A", 16))     # 42
print(base_to_decimal("400", 5))     # 100
print(base_to_decimal("73", 36))     # 255

Функция перевода между любыми системами:

def convert_base(number_str, from_base, to_base):
    """Переводит число из одной системы счисления в другую"""
    # Сначала переводим в десятичную
    decimal = base_to_decimal(number_str, from_base)
    # Затем переводим в целевую систему
    return decimal_to_base(decimal, to_base)

# Примеры
print(convert_base("101010", 2, 16))   # 2A (из двоичной в шестнадцатеричную)
print(convert_base("2A", 16, 2))       # 101010 (из шестнадцатеричной в двоичную)
print(convert_base("52", 8, 2))        # 101010 (из восьмеричной в двоичную)
print(convert_base("101010", 2, 8))    # 52 (из двоичной в восьмеричную)

4. Работа с байтами и битами

Системы счисления особенно полезны при работе с байтами и битами.

Работа с байтами:

# Представление байта в разных системах
byte_value = 170  # 10101010 в двоичной

print(f"Десятичное: {byte_value}")
print(f"Двоичное: {bin(byte_value)[2:].zfill(8)}")  # 10101010
print(f"Восьмеричное: {oct(byte_value)[2:].zfill(3)}")  # 252
print(f"Шестнадцатеричное: {hex(byte_value)[2:].upper().zfill(2)}")  # AA

# Проверка отдельных битов
def get_bit(byte, position):
    """Возвращает значение бита в позиции position (0-7)"""
    return (byte >> position) & 1

byte = 170  # 10101010
for i in range(8):
    print(f"Бит {i}: {get_bit(byte, i)}")

# Установка бита
def set_bit(byte, position, value):
    """Устанавливает бит в позиции position в значение value (0 или 1)"""
    if value:
        return byte | (1 << position)
    else:
        return byte & ~(1 << position)

byte = 170
byte = set_bit(byte, 0, 1)  # Установить младший бит в 1
print(f"После установки: {bin(byte)[2:].zfill(8)}")

5. Практические примеры

Пример 1: Калькулятор систем счисления

def number_calculator(num1_str, base1, num2_str, base2, operation, result_base):
    """Выполняет арифметическую операцию над числами в разных системах"""
    # Переводим в десятичную
    num1 = int(num1_str, base1)
    num2 = int(num2_str, base2)
    
    # Выполняем операцию
    if operation == '+':
        result = num1 + num2
    elif operation == '-':
        result = num1 - num2
    elif operation == '*':
        result = num1 * num2
    elif operation == '/':
        result = num1 // num2  # Целочисленное деление
    else:
        return "Неизвестная операция"
    
    # Переводим результат
    if result_base == 2:
        return bin(result)[2:]
    elif result_base == 8:
        return oct(result)[2:]
    elif result_base == 16:
        return hex(result)[2:].upper()
    else:
        return str(result)

# Примеры
print(number_calculator("1011", 2, "1101", 2, "+", 2))  # 11000
print(number_calculator("175", 8, "246", 8, "+", 8))   # 443
print(number_calculator("1A5", 16, "2B3", 16, "+", 16)) # 458

Пример 2: Проверка палиндрома в разных системах

def is_palindrome_in_base(n, base):
    """Проверяет, является ли число палиндромом в заданной системе счисления"""
    if base == 2:
        num_str = bin(n)[2:]
    elif base == 8:
        num_str = oct(n)[2:]
    elif base == 16:
        num_str = hex(n)[2:].upper()
    else:
        num_str = decimal_to_base(n, base)
    
    return num_str == num_str[::-1]

# Примеры
print(is_palindrome_in_base(9, 2))   # True (9 = 1001₂ - палиндром)
print(is_palindrome_in_base(17, 16)) # True (17 = 11₁₆ - палиндром)
print(is_palindrome_in_base(10, 2))  # False (10 = 1010₂ - не палиндром)

8. Типичные ошибки

Ошибка 1: Забыли убрать префикс

# НЕПРАВИЛЬНО
binary = bin(42)
print(binary)  # Выведет "0b101010" - с префиксом!

# ПРАВИЛЬНО
binary = bin(42)[2:]  # Убираем первые 2 символа
print(binary)  # Выведет "101010"

# Или используй format()
binary = format(42, 'b')
print(binary)  # "101010" (без префикса)

Ошибка 2: Не указали основание в int()

# НЕПРАВИЛЬНО
binary = "101010"
decimal = int(binary)  # Будет интерпретировано как десятичное!
print(decimal)  # 101010 (неправильно!)

# ПРАВИЛЬНО
binary = "101010"
decimal = int(binary, 2)  # Указываем основание 2
print(decimal)  # 42 (правильно!)

Ошибка 3: Использование неправильных символов

# НЕПРАВИЛЬНО
hex_num = "1G5"  # G не существует в шестнадцатеричной системе!
# int(hex_num, 16) выдаст ошибку ValueError

# ПРАВИЛЬНО
hex_num = "1A5"  # Используем только 0-9 и A-F
decimal = int(hex_num, 16)
print(decimal)  # 421

# Python не различает регистр
print(int("1a5", 16))  # 421
print(int("1A5", 16))  # 421

Ошибка 4: Путаница с форматом format()

# НЕПРАВИЛЬНО - неправильный формат
number = 42
print(format(number, 'binary'))  # Ошибка!

# ПРАВИЛЬНО - используй правильные коды формата
number = 42
print(format(number, 'b'))   # Двоичная
print(format(number, 'o'))   # Восьмеричная
print(format(number, 'x'))   # Шестнадцатеричная (строчные)
print(format(number, 'X'))   # Шестнадцатеричная (заглавные)
print(format(number, 'd'))   # Десятичная (по умолчанию)

10. Резюме урока

  • bin() - перевод в двоичную систему (возвращает строку с префиксом 0b)
  • oct() - перевод в восьмеричную систему (возвращает строку с префиксом 0o)
  • hex() - перевод в шестнадцатеричную систему (возвращает строку с префиксом 0x)
  • int(строка, основание) - перевод из любой системы в десятичную
  • format(число, код) - форматирование: 'b' (двоичная), 'o' (восьмеричная), 'x'/'X' (шестнадцатеричная)
  • Префиксы - всегда убирай [2:] при использовании bin(), oct(), hex()
  • Основание в int() - всегда указывай вторым аргументом (2, 8, 16 и т.д.)
  • Регистр букв - в шестнадцатеричной системе не важен для int(), но важен для format()
  • Собственные функции - для систем с основаниями, отличными от 2, 8, 16, нужно создавать свои функции
  • Форматирование с шириной - '08b' означает 8 символов с ведущими нулями в двоичной системе
