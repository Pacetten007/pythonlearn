1. Поиск подстроки в строке

Одна из самых частых задач - найти, содержится ли одна строка в другой.

1.1. Простой поиск (встроенный метод)

Пример: Использование оператора in

text = "Привет, мир!"
substring = "мир"

# Простой способ
if substring in text:
    print(f"'{substring}' найдено в строке")
    index = text.find(substring)  # Находит индекс первого вхождения
    print(f"Индекс: {index}")
else:
    print(f"'{substring}' не найдено")

# Метод find() возвращает -1, если не найдено
index = text.find("Python")
if index != -1:
    print(f"Найдено на позиции {index}")
else:
    print("Не найдено")

1.2. Поиск всех вхождений

Пример: Найти все вхождения подстроки

def find_all_occurrences(text, substring):
    """Находит все индексы, где встречается подстрока"""
    indices = []
    start = 0
    while True:
        index = text.find(substring, start)
        if index == -1:
            break
        indices.append(index)
        start = index + 1
    return indices

# Пример
text = "банан банан банан"
substring = "банан"
indices = find_all_occurrences(text, substring)
print(f"'{substring}' найдено на позициях: {indices}")

1.3. Поиск подстроки вручную (алгоритм)

Пример: Реализация поиска без встроенных методов

def find_substring(text, substring):
    """Находит первое вхождение подстроки вручную"""
    n = len(text)
    m = len(substring)
    
    if m == 0:
        return 0
    if m > n:
        return -1
    
    for i in range(n - m + 1):
        # Проверяем, совпадает ли подстрока начиная с позиции i
        match = True
        for j in range(m):
            if text[i + j] != substring[j]:
                match = False
                break
        if match:
            return i
    return -1

# Пример
text = "программирование"
substring = "грам"
index = find_substring(text, substring)
print(f"'{substring}' найдено на позиции: {index}")

2. Подсчет символов и слов

Часто нужно посчитать, сколько раз встречается определенный символ или слово.

2.1. Подсчет символов

Пример: Подсчет символов

def count_char(text, char):
    """Подсчитывает количество вхождений символа"""
    count = 0
    for c in text:
        if c == char:
            count += 1
    return count

# Или используя метод count()
text = "программирование"
count_a = text.count('а')
print(f"Буква 'а' встречается {count_a} раз")

# Подсчет всех символов
def count_all_chars(text):
    """Подсчитывает частоту каждого символа"""
    char_count = {}
    for char in text:
        if char in char_count:
            char_count[char] += 1
        else:
            char_count[char] = 1
    return char_count

text = "привет"
char_freq = count_all_chars(text)
print(char_freq)  # {'п': 1, 'р': 1, 'и': 1, 'в': 1, 'е': 1, 'т': 1}

2.2. Подсчет слов

Пример: Подсчет слов в строке

def count_words(text):
    """Подсчитывает количество слов в строке"""
    words = text.split()  # Разделяет по пробелам
    return len(words)

# Пример
text = "Python это язык программирования"
word_count = count_words(text)
print(f"Количество слов: {word_count}")

# Подсчет конкретного слова
def count_word_occurrences(text, word):
    """Подсчитывает количество вхождений слова"""
    words = text.split()
    count = 0
    for w in words:
        if w.lower() == word.lower():  # Без учета регистра
            count += 1
    return count

text = "кот собака кот мышь кот"
count = count_word_occurrences(text, "кот")
print(f"Слово 'кот' встречается {count} раз")

3. Проверка на палиндром (строковый)

Палиндром - это строка, которая читается одинаково слева направо и справа налево (например, "топот", "казак", "а роза упала на лапу Азора").

Пример: Проверка на палиндром

def is_palindrome(text):
    """Проверяет, является ли строка палиндромом"""
    # Убираем пробелы и приводим к нижнему регистру
    cleaned = text.replace(" ", "").lower()
    # Сравниваем строку с перевернутой
    return cleaned == cleaned[::-1]

# Примеры
print(is_palindrome("топот"))      # True
print(is_palindrome("казак"))      # True
print(is_palindrome("привет"))     # False
print(is_palindrome("А роза упала на лапу Азора"))  # True

Альтернативный способ (посимвольное сравнение)

def is_palindrome_manual(text):
    """Проверка палиндрома посимвольно"""
    cleaned = text.replace(" ", "").lower()
    n = len(cleaned)
    for i in range(n // 2):
        if cleaned[i] != cleaned[n - 1 - i]:
            return False
    return True

print(is_palindrome_manual("топот"))  # True

4. Переворот строки

Переворот строки - создание строки, в которой символы идут в обратном порядке.

Пример: Переворот строки

def reverse_string(text):
    """Переворачивает строку"""
    return text[::-1]

# Или вручную
def reverse_string_manual(text):
    """Переворачивает строку вручную"""
    reversed_text = ""
    for i in range(len(text) - 1, -1, -1):
        reversed_text += text[i]
    return reversed_text

# Примеры
text = "привет"
reversed_text = reverse_string(text)
print(f"'{text}' → '{reversed_text}'")  # 'привет' → 'тевирп'

5. Удаление пробелов и лишних символов

Часто нужно очистить строку от лишних пробелов или символов.

Пример: Удаление пробелов

text = "  привет  мир  "

# Удаление пробелов с начала и конца
trimmed = text.strip()
print(f"'{trimmed}'")  # 'привет  мир'

# Удаление всех пробелов
no_spaces = text.replace(" ", "")
print(f"'{no_spaces}'")  # 'приветмир'

# Удаление пробелов с начала
left_trimmed = text.lstrip()
print(f"'{left_trimmed}'")  # 'привет  мир  '

# Удаление пробелов с конца
right_trimmed = text.rstrip()
print(f"'{right_trimmed}'")  # '  привет  мир'

Пример: Удаление всех пробелов вручную

def remove_spaces(text):
    """Удаляет все пробелы из строки"""
    result = ""
    for char in text:
        if char != " ":
            result += char
    return result

text = "привет мир Python"
cleaned = remove_spaces(text)
print(f"'{text}' → '{cleaned}'")

6. Замена символов и подстрок

Замена - это изменение части строки на другую.

Пример: Замена символов

text = "привет мир"

# Замена всех вхождений
new_text = text.replace("мир", "Python")
print(new_text)  # "привет Python"

# Замена только первого вхождения
def replace_first(text, old, new):
    """Заменяет только первое вхождение"""
    index = text.find(old)
    if index == -1:
        return text
    return text[:index] + new + text[index + len(old):]

text = "кот кот кот"
new_text = replace_first(text, "кот", "собака")
print(new_text)  # "собака кот кот"

7. Разделение и объединение строк

Разделение строки на части и объединение частей в одну строку.

Пример: Разделение строки

text = "яблоко,банан,апельсин"

# Разделение по запятой
fruits = text.split(",")
print(fruits)  # ['яблоко', 'банан', 'апельсин']

# Разделение по пробелам
text = "привет мир Python"
words = text.split()
print(words)  # ['привет', 'мир', 'Python']

# Разделение с ограничением количества частей
text = "один два три четыре"
parts = text.split(" ", 2)  # Максимум 2 разделения
print(parts)  # ['один', 'два', 'три четыре']

Пример: Объединение строк

words = ["привет", "мир", "Python"]

# Объединение через пробел
sentence = " ".join(words)
print(sentence)  # "привет мир Python"

# Объединение через запятую
fruits = ["яблоко", "банан", "апельсин"]
fruits_str = ", ".join(fruits)
print(fruits_str)  # "яблоко, банан, апельсин"

# Объединение без разделителя
text = "".join(words)
print(text)  # "приветмирPython"

10. Типичные ошибки

Ошибка 1: Изменение строки напрямую

# НЕПРАВИЛЬНО - строки неизменяемы!
text = "привет"
text[0] = "П"  # Ошибка! TypeError

# ПРАВИЛЬНО - создай новую строку
text = "привет"
new_text = "П" + text[1:]
print(new_text)  # "Привет"

Ошибка 2: Неправильное сравнение строк с учетом регистра

# НЕПРАВИЛЬНО - регистр важен
text1 = "Привет"
text2 = "привет"
if text1 == text2:  # False!
    print("Равны")

# ПРАВИЛЬНО - приведи к одному регистру
text1 = "Привет"
text2 = "привет"
if text1.lower() == text2.lower():
    print("Равны")  # Выполнится

Ошибка 3: Забыли обработать пустую строку

# НЕПРАВИЛЬНО - не обработан случай пустой строки
def first_char(text):
    return text[0]  # Ошибка, если text пустая!

# ПРАВИЛЬНО - проверь длину
def first_char(text):
    if len(text) == 0:
        return None
    return text[0]

Ошибка 4: Неэффективное конкатенирование строк в цикле

# НЕПРАВИЛЬНО - медленно для больших строк
result = ""
for i in range(1000):
    result += "a"  # Каждый раз создается новая строка!

# ПРАВИЛЬНО - используй список и join()
result_list = []
for i in range(1000):
    result_list.append("a")
result = "".join(result_list)  # Быстро!

12. Резюме урока

  • Поиск подстроки - используй in, find() или реализуй алгоритм вручную
  • Подсчет символов - используй цикл или метод count()
  • Палиндром - строка, которая читается одинаково в обоих направлениях
  • Переворот строки - используй срез [::-1]
  • Разделение строки - метод split()
  • Объединение строк - метод join()
  • Строки неизменяемы - нельзя изменить символ напрямую, нужно создавать новую строку
  • Для эффективной конкатенации используй список и join()
  • Всегда обрабатывай граничные случаи (пустая строка, отсутствие подстроки)
